# Compute binary-classification metrics from predicted probabilities
evaluate_model <- function(prob, truth, levels = c("HRP", "HRD")) {
  truth <- factor(truth, levels = levels)

  # ROC curve and its 95% confidence interval (DeLong method)
  roc_obj <- pROC::roc(truth, prob,
                       levels = levels, direction = "<", quiet = TRUE)
  auc_val <- as.numeric(pROC::auc(roc_obj))
  ci_val  <- as.numeric(pROC::ci.auc(roc_obj, method = "delong"))

  # Optimal threshold determined by the Youden index
  youden   <- roc_obj$sensitivities + roc_obj$specificities - 1
  best_thr <- roc_obj$thresholds[which.max(youden)]

  pred_cls <- factor(ifelse(prob > best_thr, "HRD", "HRP"), levels = levels)
  cm <- caret::confusionMatrix(pred_cls, truth, positive = "HRD")

  list(
    AUC         = auc_val,
    CI_lower    = ci_val[1],
    CI_upper    = ci_val[3],
    Sensitivity = as.numeric(cm$byClass["Sensitivity"]),
    Specificity = as.numeric(cm$byClass["Specificity"]),
    Accuracy    = as.numeric(cm$overall["Accuracy"]),
    F1          = as.numeric(cm$byClass["F1"]),
    Threshold   = best_thr,
    ROC         = roc_obj,
    Confusion   = cm
  )
}

# Format a single evaluation result as one summary row
summary_row <- function(res, dataset, model_name) {
  data.frame(
    Dataset     = dataset,
    Model       = model_name,
    AUC         = round(res$AUC, 4),
    CI_95       = paste0("(", round(res$CI_lower, 4),
                         "-",  round(res$CI_upper, 4), ")"),
    Sensitivity = round(res$Sensitivity, 4),
    Specificity = round(res$Specificity, 4),
    Accuracy    = round(res$Accuracy, 4),
    F1          = round(res$F1, 4),
    stringsAsFactors = FALSE
  )
}


# =============================================================================
# 2. Load TCGA data and build label vector
# =============================================================================
# Expression matrix: rows = genes, columns = samples.
# The first 134 columns are HRP and the remaining 159 are HRD.

tcga_file <- "D:/PRJs/prj_OV_HRD/OV134-HRP-159-HRD.txt"
expr_tcga <- as.matrix(read.table(tcga_file, header = TRUE, row.names = 1,
                                  sep = "\t", check.names = FALSE))
cat("TCGA expression matrix:", dim(expr_tcga), "\n")

sample_ids <- colnames(expr_tcga)
group_tcga <- factor(c(rep("HRP", 134), rep("HRD", 159)),
                     levels = c("HRP", "HRD"))
names(group_tcga) <- sample_ids


# =============================================================================
# 3. Stratified 70/30 split into training and test sets
# =============================================================================

set.seed(2026)
train_index   <- createDataPartition(group_tcga, p = 0.7, list = FALSE)
train_samples <- sample_ids[train_index]
test_samples  <- setdiff(sample_ids, train_samples)

expr_train  <- expr_tcga[, train_samples, drop = FALSE]
expr_test   <- expr_tcga[, test_samples,  drop = FALSE]
group_train <- group_tcga[train_samples]
group_test  <- group_tcga[test_samples]

cat("Training set:", ncol(expr_train),
    " (HRP:", sum(group_train == "HRP"),
    ", HRD:", sum(group_train == "HRD"), ")\n")
cat("Test set:    ", ncol(expr_test),
    " (HRP:", sum(group_test  == "HRP"),
    ", HRD:", sum(group_test  == "HRD"), ")\n")


# =============================================================================
# 4. Differential expression analysis on the training set (limma)
# =============================================================================

design <- model.matrix(~ group_train)
colnames(design) <- c("(Intercept)", "HRDvsHRP")

fit         <- eBayes(lmFit(expr_train, design))
deg_results <- topTable(fit, coef = "HRDvsHRP",
                        number = Inf, sort.by = "none")

adj_p_cutoff <- 0.05
logfc_cutoff <- 1

deg_genes <- rownames(deg_results)[
  deg_results$adj.P.Val < adj_p_cutoff & abs(deg_results$logFC) > logfc_cutoff
]
cat("Number of DEGs:", length(deg_genes), "\n")

write.csv(deg_results,                 "DEGs_train_limma_full.csv")
write.csv(deg_results[deg_genes, ],    "DEGs_train_limma_filtered.csv")


# =============================================================================
# 5. Boruta feature selection on the training set
# =============================================================================

set.seed(456)
X_boruta <- t(expr_train[deg_genes, , drop = FALSE])
y_boruta <- group_train

boruta_result    <- Boruta(X_boruta, y_boruta, doTrace = 1, maxRuns = 200)
boruta_important <- getSelectedAttributes(boruta_result, withTentative = FALSE)
cat("Boruta-confirmed genes:", length(boruta_important), "\n")

selected_genes <- if (length(boruta_important) > 0) boruta_important else deg_genes
write.csv(data.frame(gene = selected_genes),
          "Boruta_genes.csv", row.names = FALSE)


# =============================================================================
# 6. LASSO-based stability selection (100 repeats x 10-fold CV)
# =============================================================================

X_sel        <- t(expr_train[selected_genes, , drop = FALSE])
X_sel_scaled <- scale(X_sel)

set.seed(789)
n_rep  <- 100
nfolds <- 10
freq   <- setNames(rep(0, length(selected_genes)), selected_genes)

for (i in seq_len(n_rep)) {
  foldid <- sample(rep(1:nfolds, length.out = nrow(X_sel_scaled)))
  ok <- all(sapply(1:nfolds, function(k)
    length(unique(y_boruta[foldid == k])) > 1))
  if (!ok) next

  cv_fit <- cv.glmnet(X_sel_scaled, y_boruta,
                      family = "binomial", alpha = 1,
                      type.measure = "auc",
                      foldid = foldid, nfolds = nfolds)

  coef_min <- coef(cv_fit, s = "lambda.min")[-1, , drop = FALSE]
  sel      <- rownames(coef_min)[coef_min[, 1] != 0]
  freq[sel] <- freq[sel] + 1
}

freq_df <- data.frame(gene = names(freq),
                      freq = freq,
                      prop = freq / n_rep) %>%
  arrange(desc(prop))
write.csv(freq_df, "stability_selection_frequencies.csv", row.names = FALSE)

threshold    <- 0.7
stable_genes <- freq_df$gene[freq_df$prop >= threshold]
if (length(stable_genes) < 3) {
  warning("Too few stable genes; selecting top-10 by frequency instead.")
  stable_genes <- freq_df$gene[1:min(10, nrow(freq_df))]
}
cat("Final stable gene set (n =", length(stable_genes), ")\n")
write.csv(data.frame(gene = stable_genes),
          "stable_genes_selected.csv", row.names = FALSE)

# One additional LASSO run on the full training set for the trajectory plot
set.seed(789)
cv_lasso_final <- cv.glmnet(X_sel_scaled, y_boruta,
                            family = "binomial", alpha = 1,
                            nfolds = 10, type.measure = "deviance")


# =============================================================================
# 7. Build the final modeling matrices (sample x gene)
# =============================================================================

# Training set
X_fin_train_raw <- t(expr_train[stable_genes, , drop = FALSE])
colnames(X_fin_train_raw) <- stable_genes

# Test set
X_fin_test_raw <- t(expr_test[stable_genes, , drop = FALSE])
colnames(X_fin_test_raw) <- stable_genes

# Standardization using training-set mean/SD (applied identically to test)
train_mean <- apply(X_fin_train_raw, 2, mean)
train_sd   <- apply(X_fin_train_raw, 2, sd)
train_sd[train_sd == 0] <- 1

X_fin_train <- as.data.frame(
  sweep(sweep(X_fin_train_raw, 2, train_mean, "-"), 2, train_sd, "/"))
X_fin_test  <- as.data.frame(
  sweep(sweep(X_fin_test_raw,  2, train_mean, "-"), 2, train_sd, "/"))

# Attach class labels
X_fin_train$Class <- group_train
X_fin_test$Class  <- group_test


# =============================================================================
# 8. Train six classifiers with 10-fold cross-validation
# =============================================================================

ctrl <- trainControl(
  method          = "cv",
  number          = 10,
  classProbs      = TRUE,
  summaryFunction = twoClassSummary,
  savePredictions = "final",
  allowParallel   = FALSE,
  verboseIter     = FALSE
)

methods <- list(
  XGBoost = "xgbTree",
  NNET    = "nnet",
  RF      = "rf",
  KKNN    = "kknn",
  DT      = "rpart",
  NB      = "nb"
)

model_list <- list()
cv_results <- list()

for (m in names(methods)) {
  cat("\n>>> Training model:", m, "...\n")
  set.seed(2024)
  fit_m <- tryCatch({
    caret::train(Class ~ ., data = X_fin_train,
                 method     = methods[[m]],
                 trControl  = ctrl,
                 metric     = "ROC",
                 tuneLength = 5,
                 preProcess = NULL)
  }, error = function(e) {
    cat("Model", m, "failed:", conditionMessage(e), "\n"); NULL
  })
  if (is.null(fit_m)) next

  model_list[[m]] <- fit_m

  cv_res <- evaluate_model(prob = fit_m$pred$HRD, truth = fit_m$pred$obs)
  cv_results[[m]] <- cv_res
  cat(sprintf("  CV AUC = %.4f\n", cv_res$AUC))
}


# =============================================================================
# 9. Select the best model by cross-validated AUC
# =============================================================================

cv_auc_df <- data.frame(
  Model  = names(cv_results),
  CV_AUC = sapply(cv_results, function(x) x$AUC)
) %>% arrange(desc(CV_AUC))

print(cv_auc_df)

best_method <- cv_auc_df$Model[1]
best_model  <- model_list[[best_method]]
cat(sprintf("\n>>> Best model: %s (CV AUC = %.4f)\n",
            best_method, cv_auc_df$CV_AUC[1]))


# =============================================================================
# 10. Evaluate the best model on the TCGA test set
# =============================================================================

prob_test <- predict(best_model, newdata = X_fin_test, type = "prob")[, "HRD"]
res_test  <- evaluate_model(prob_test, group_test)


# =============================================================================
# 11. External validation on the AOCS cohort
# =============================================================================
# Expression matrix: rows = genes, columns = samples (log2 scale)
# Metadata file: first column = sample ID, additional column "HRD.group"

aocs_file      <- "D:/PRJs/prj_OV_HRD/ICGC/GELL_ALL_UNIQ.txt"
aocs_meta_file <- "D:/PRJs/prj_OV_HRD/ICGC/group.txt"

res_aocs <- NULL
if (file.exists(aocs_file) && file.exists(aocs_meta_file)) {

  # --- Load expression and metadata (header = TRUE is critical) ---
  expr_aocs <- as.matrix(read.table(aocs_file, header = TRUE, row.names = 1,
                                    sep = "\t", check.names = FALSE))
  meta_aocs <- read.table(aocs_meta_file, row.names = 1, header = TRUE)

  cat("AOCS expression matrix:", dim(expr_aocs), "\n")
  cat("AOCS metadata columns:  ",
      paste(colnames(meta_aocs), collapse = ", "), "\n")

  # --- Match samples ---
  common_samples <- intersect(colnames(expr_aocs), rownames(meta_aocs))
  stopifnot(length(common_samples) > 0)
  expr_aocs <- expr_aocs[, common_samples, drop = FALSE]
  meta_aocs <- meta_aocs[common_samples, , drop = FALSE]

  # --- Extract labels ----
  if (!"HRD.group" %in% colnames(meta_aocs)) {
    stop("Column 'HRD.group' not found in AOCS metadata.")
  }
  group_aocs <- factor(meta_aocs$HRD.group, levels = c("HRP", "HRD"))
  cat("AOCS label distribution:\n"); print(table(group_aocs))

  # --- Restrict to the stable genes retained in the model ---
  genes_avail <- intersect(stable_genes, rownames(expr_aocs))
  missing_g   <- setdiff(stable_genes, genes_avail)
  if (length(missing_g) > 0) {
    warning("Genes missing from AOCS: ", paste(missing_g, collapse = ", "))
  }

  # --- Standardize using the training-set mean and SD ---
  X_aocs_raw <- t(expr_aocs[genes_avail, , drop = FALSE])
  colnames(X_aocs_raw) <- genes_avail
  X_aocs <- as.data.frame(
    sweep(sweep(X_aocs_raw, 2, train_mean[genes_avail], "-"),
          2, train_sd[genes_avail], "/"))

  # Missing genes are imputed with 0 (which equals the training mean)
  for (g in stable_genes) {
    if (!g %in% colnames(X_aocs)) X_aocs[[g]] <- 0
  }
  X_aocs <- X_aocs[, stable_genes, drop = FALSE]

  stopifnot(nrow(X_aocs) == length(group_aocs))
  X_aocs$Class <- group_aocs

  # --- Predict and evaluate ---
  prob_aocs <- predict(best_model, newdata = X_aocs, type = "prob")[, "HRD"]
  res_aocs  <- evaluate_model(prob_aocs, group_aocs)

  cat("\nAOCS standardized mean per gene (should be near 0):\n")
  print(round(colMeans(X_aocs[, stable_genes, drop = FALSE]), 3))

} else {
  cat("AOCS files not found; skipping external validation.\n")
}

# Helper: extract the CV out-of-fold ROC at the best hyperparameters
get_cv_roc <- function(fit) {
  pred <- fit$pred
  bt   <- fit$bestTune
  if (!is.null(bt) && ncol(bt) > 0) {
    keep <- rep(TRUE, nrow(pred))
    for (col in colnames(bt)) keep <- keep & (pred[[col]] == bt[[col]])
    pred <- pred[keep, ]
  }
  pROC::roc(pred$obs, pred$HRD, levels = c("HRP", "HRD"),
            direction = "<", quiet = TRUE)
}

}

# Plotting functions
plot_multi_roc <- function(roc_list, title, cex_leg = 0.85) {
  ms   <- names(roc_list)
  cols <- model_colors[ms]
  plot(roc_list[[1]], col = cols[1], lwd = 2, legacy.axes = TRUE,
       main = title, cex.main = 1.3, font.main = 2)
  if (length(roc_list) > 1) {
    for (i in 2:length(roc_list))
      plot(roc_list[[i]], col = cols[i], lwd = 2, add = TRUE)
  }
  abline(a = 0, b = 1, lty = 3, col = "grey60")
  labs <- sprintf("%s (AUC=%.3f)", ms,
                  sapply(roc_list, function(x) as.numeric(pROC::auc(x))))
  legend("bottomright", legend = labs, col = cols, lwd = 2,
         cex = cex_leg, bty = "n")
}

plot_single_roc <- function(roc_obj, title, col = "#A65628",
                            label = NULL, cex_leg = 1.0) {
  plot(roc_obj, col = col, lwd = 2.2, legacy.axes = TRUE,
       main = title, cex.main = 1.3, font.main = 2)
  abline(a = 0, b = 1, lty = 3, col = "grey60")
  if (is.null(label))
    label <- sprintf("AUC = %.3f", as.numeric(pROC::auc(roc_obj)))
  legend("bottomright", legend = label, col = col, lwd = 2,
         cex = cex_leg, bty = "n")
}

# ---- 17.2 Which signature genes are available in the TCGA data? -----------
sig_in_tcga  <- intersect(signature_217, rownames(expr_tcga))

      paste(sig_missing, collapse = ", "), "\n")

# ---- 17.3 Build feature matrices (standardization identical to our model) --
# Training set
X_sig_train_raw <- t(expr_train[sig_in_tcga, , drop = FALSE])
colnames(X_sig_train_raw) <- sig_in_tcga

# Test set
X_sig_test_raw <- t(expr_test[sig_in_tcga, , drop = FALSE])
colnames(X_sig_test_raw) <- sig_in_tcga

# Standardize using TRAINING-set mean and SD (same policy as our main model)
sig_mean <- apply(X_sig_train_raw, 2, mean)
sig_sd   <- apply(X_sig_train_raw, 2, sd)
sig_sd[sig_sd == 0] <- 1

X_sig_train <- as.data.frame(
  sweep(sweep(X_sig_train_raw, 2, sig_mean, "-"), 2, sig_sd, "/"))
X_sig_test <- as.data.frame(
  sweep(sweep(X_sig_test_raw, 2, sig_mean, "-"), 2, sig_sd, "/"))

X_sig_train$Class <- group_train
X_sig_test$Class  <- group_test


# ---- 17.4 Fit the SAME model family as our best model (fair framework) -----
# Using the identical model class as `best_model` ensures the ONLY difference
# under test is the feature set (our stable genes vs. the published 217 genes).
set.seed(2024)

sig_ctrl <- trainControl(
  method          = "cv",
  number          = 10,
  classProbs      = TRUE,
  summaryFunction = twoClassSummary,
  savePredictions = "final",
  allowParallel   = FALSE,
  verboseIter     = FALSE
)

sig_model <- caret::train(
  Class ~ .,
  data       = X_sig_train,
  method     = methods[[best_method]],   # = "nb"
  trControl  = sig_ctrl,
  metric     = "ROC",
  tuneLength = 5,
  preProcess = NULL
)

cat(sprintf("\nExternal signature model trained (method = %s).\n",
            best_method))


# ---- 17.5 Cross-validated performance on the training set -----------------
cv_sig <- evaluate_model(prob = sig_model$pred$HRD,
                         truth = sig_model$pred$obs)
cat(sprintf("[Signature-217]  Train  10-fold CV AUC = %.4f\n", cv_sig$AUC))


# ---- 17.6 TCGA test set ---------------------------------------------------

res_sig_test  <- evaluate_model(prob_sig_test, group_test)
cat(sprintf("[Signature-217]  TCGA test AUC = %.4f\n", res_sig_test$AUC))

  X_sig_aocs <- as.data.frame(
    sweep(sweep(X_sig_aocs_raw, 2, sig_mean[sig_in_aocs], "-"),
          2, sig_sd[sig_in_aocs], "/"))
